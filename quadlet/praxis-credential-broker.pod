[Unit]
Description=Praxis credential broker pod
StartLimitIntervalSec=0
Wants=network-online.target
After=network-online.target

[Pod]
PodName=praxis-credential-broker
ExitPolicy=continue
PublishPort=127.0.0.1:18080:8081
PublishPort=100.121.0.115:18080:8081
PublishPort=127.0.0.1:18083:8082
PublishPort=100.121.0.115:18083:8082

[Service]
# These commands run on Xenon's host, before Podman binds the tailnet address.
# Do not replace them with a tailscaled.service dependency: user services and
# the system Tailscale daemon do not have a safe ordering relationship here.
ExecStartPre=/usr/bin/tailscale wait --timeout=30s
ExecStartPre=/usr/bin/tailscale ip --4 --assert=100.121.0.115
Restart=on-failure
RestartSec=15s
TimeoutStartSec=120s

[Install]
WantedBy=default.target
